/**

 *

 *    于是引述内容顶到了 `$` 锚点上 —— **转述别人的话会真出图**：
 *      他说“出图吧” / 老板说“直接出图吧” / 同事说“做张图”  **false → true**

const TRAILING_DECOR = /(?:[\s…~～。．]|\p{Extended_Pictographic}|\uFE0F|[\u{1F3FB}-\u{1F3FF}])*$/u;

/**
 * 分句。
 *
 * ⚠️ 分隔符必须包含 ASCII `.`、破折号、冒号、省略号 —— 评审 High-3：第二版只切
 *    `。！？!?；;,，、\n`，于是 `go ahead. no wait.` / `go ahead — no wait` /
 *    `Render it. Actually no.` 里的反悔全部看不见。上一轮那 4 个例子恰好都用了 `,` 或 `？`，
 *    语料对这个缺口同样是结构性失明。
 *    `.` 要求后接空白或行尾，别切坏 `3.5cm`。
 */
function clauses(s: string): Clause[] {
  const parts = s.split(/(?<=[。！？!?])|[；;,，、\n]+|(?<=\.)(?=\s)|\s*[—–…:：]+\s*/u);
  const out: Clause[] = [];
  for (const raw of parts) {
    const t = (raw ?? '').trim();
    if (!t) continue;
    const m = /[。！？!?.]$/.exec(t);
    const end = m ? m[0] : '';
    const body = end ? t.slice(0, -1) : t;
    const core = body.replace(TRAILING_DECOR, '').trim();
    if (core) out.push({ core, end });
  }
  return out;
}

/** 指令分句自身的否定（「我不想出图」「没必要出图」）。裸「不」排除「不错/不妨」（肯定义）。 */
const NEGATORS = /(?:\bnot\b|n['’]t\b|\bnever\b|\bno\s+need\b|\bwithout\b|\bavoid\b|别|不要|不用|先不|暂不|暂时不|无需|无须|还不|先别|算了|没(?:必要|打算)|不(?!错|妨))/i;

/**
 * 疑问判定。
 *
 * ⚠️ `will` **必须在表里** —— 评审 High-2：第二版重写时把它弄丢了，于是
 *    `Will you go ahead and render it` **从 false 变成 true**，一句礼貌的英文问句直接烧一次 GPU。
 *    这条正踩在本文件声明要保护的那个方向上，前面三轮都没抓到。
 * ⚠️ 「吧」**不算**疑问语气词 —— 「就这个吧」「出图吧」是同意/建议。
 */
const INTERROGATIVE_LEAD = /^(?:do|does|did|should|shall|can|could|would|will|won['’]?t|wanna|is|are|am|may|might|what|how|why|when|which|who|whether)\b|^(?:要不要|是否|能不能|可不可以|行不行|好不好|该不该)/i;
const INTERROGATIVE_ANY = /(?:什么|多久|多少|怎么|哪个|哪种|几天|几个|贵不贵|收费)/;
function isQuestion(c: Clause): boolean {
  if (/[?？]/.test(c.end)) return true;
  if (/[吗呢么]$/.test(c.core)) return true;
  if (INTERROGATIVE_LEAD.test(c.core)) return true;
  return INTERROGATIVE_ANY.test(c.core);
}

/**
 * 客户这句话是不是一条明确的「开干」指令。
 *
 * ⚠️ 超长输入取**尾部** 2000 字符（指令的语义在句尾）。已知边界：单个分句超 2000 字符时
 *    会砍掉分句开头的否定词（评审 Low，很牵强但方向是烧钱那边）。
 */
export function hasExplicitInstruction(text: string): boolean {
  const raw = (text ?? '').trim();
  if (!raw) return false;
  const t = raw.length > 2000 ? raw.slice(-2000) : raw;

  const cs = clauses(t);
  let i = -1;
  let matchEnd = 0;
  for (let k = cs.length - 1; k >= 0; k--) {
    const hit = GO_PATTERNS.map((re) => re.exec(cs[k]!.core)).find(Boolean);
    if (hit) { i = k; matchEnd = hit.index + hit[0].length; break; }
  }
  if (i < 0) return false;

  const c = cs[i]!;
  // 🔴 修饰语作用域:整条消息里有条件/推迟/转述/征求许可/犹豫 → 不是「现在就干」。
  //    放在最前面,因为它们【独立于位置】,前置后置都算 —— 这正是锚点表达不了的那部分。
  if (modifierHit(t)) return false;
  if (NEGATORS.test(c.core)) return false;
  if (isQuestion(c)) return false;
  // 同一分句里、触发词【之后】的尾巴 —— `go ahead no wait` 完全没有标点,只能这样抓。
  if (isCancel(c.core.slice(matchEnd))) return false;
  for (let k = i + 1; k < cs.length; k++) {
    if (isCancel(cs[k]!.core)) return false;
    // 指令之后又冒出疑问 —— 「做个效果图吧，多少钱」和「做个图大概多少钱」是同一个意思,
    //   不该给出相反的花钱决定(评审 Med)。
    if (isQuestion(cs[k]!)) return false;
  }
  return true;
}

 *    只认点 chip 则首轮全灭。所以这里不是「调正则」,是把【方向确认】和【出图指令】

 *
 * 判据(全部要满足):
 *   1. 某个分句里出现盒型词,且该词处在【盒型语境】(它自己是名词短语的一部分,不是 shipping/booklet 这种词内碎片)
 *   2. 该分句没有否定词
 *   3. 该分句不是疑问句(「What is the shipping cost?」不是在选方向)
 */
const BOX_WORDS: Array<[RegExp, BoxType]> = [
  // ⚠️ 英文侧一律带词边界 —— 这正是裸 includes 的病根:`notebook`/`facebook` 里的 `book`、
  //    `shipping` 里的 `ship`、`solid`/`holiday` 里的 `lid` 都不该算。

  //    所以每条都是 `(?:\b英文\b)|中文` 的形状,别图省事合成一个大括号。
  [/(?:\b(?:lid[-\s]?(?:and[-\s]?)?base|lid\s+box|rigid\s+box)\b)|天地盒|天地盖|上下盖/i, 'lidbase'],
  [/(?:\b(?:magnet(?:ic)?(?:\s+closure)?(?:\s+box)?|book[-\s]?style\s+box)\b)|磁扣|磁吸|书型盒/i, 'magnet'],
  [/(?:\b(?:drawer(?:\s+box)?|slide[-\s]?out\s+box|sleeve\s+box)\b)|抽屉盒|抽拉盒/i, 'drawer'],
  [/(?:\b(?:mailer(?:\s+box)?|shipping\s+box|corrugated(?:\s+box)?)\b)|飞机盒|快递盒|瓦楞盒/i, 'mailer'],
];
export function statedDirection(text: string): BoxType | null {
  const raw = (text ?? '').trim();
  if (!raw) return null;
  const t = raw.length > 2000 ? raw.slice(-2000) : raw;
  for (const c of clauses(t)) {
    if (NEGATORS.test(c.core)) continue;     // 「I do NOT want a book box」
    if (isQuestion(c)) continue;             // 「What is the shipping cost?」
    for (const [re, bt] of BOX_WORDS) if (re.test(c.core)) return bt;
  }
  return null;
}

  if (slotsComplete(input.slots) && input.confirmedDirection) {
    return { allow: true, reason: 'slots-complete' };
  }
  return { allow: false, reason: 'slots-incomplete' };
}
