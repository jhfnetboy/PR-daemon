/**
 * 重编码成够小的 JPEG。
 *
 * 原图是 880×880 的 PNG（`setPixelRatio(min(dpr,2))` × `SZ=440`）——
 * 四张带光影的这种图 base64 之后就撑爆了上限，而超限是**静默丢快照**，
 * 表现成「分享页没有图」（评审抓到：功能实际是哑的）。
 *
 * ⚠️ 先铺白底再画 —— 离屏 renderer 是 `alpha:true`，直接转 JPEG 会得到黑底。
 */
function shrink(dataUrl: string, maxW: number, quality: number): Promise<string | null> {
  return new Promise((res) => {
    const im = new Image();
    im.onload = () => {
      const scale = Math.min(1, maxW / im.width);
      const cv = document.createElement('canvas');
      cv.width = Math.max(1, Math.round(im.width * scale));
      cv.height = Math.max(1, Math.round(im.height * scale));
      const g = cv.getContext('2d');
      if (!g) { res(null); return; }
      g.fillStyle = '#ffffff'; g.fillRect(0, 0, cv.width, cv.height);
      g.drawImage(im, 0, 0, cv.width, cv.height);
      const out = cv.toDataURL('image/jpeg', quality);
      res(out.length <= BLOCK_SRC_MAX ? out : null);
    };
    im.onerror = () => res(null);
    im.src = dataUrl;
  });
}

/** 分享快照里的一块。**只有图片** —— 文字规格走 `design` 白名单那条路。 */
export interface ShareBlock { t: 'image'; src: string; alt: string; cap?: string }

/**
 * 快照块的白名单。和 `pickDesign()` 同一个方向：**默认拒绝**。
 *
 * 认不出来的 `t`、不是 `data:image/...` 的 `src`、超长的说明文字 —— 一律丢掉，
 * 不是「清洗后放行」。丢掉的后果是分享页少一张图（看得出来），
 * 放行的后果是价格外流（没人会发现）。代价不对称。
 */
export function pickBlocks(raw: unknown): ShareBlock[] {
  if (!Array.isArray(raw)) return [];
  const out: ShareBlock[] = [];
  for (const item of raw) {
    if (out.length >= SHARE_BLOCKS_MAX) break;
    if (!item || typeof item !== 'object') continue;
    const b = item as Record<string, unknown>;
    if (b.t !== 'image') continue;
    const src = typeof b.src === 'string' ? b.src : '';
    if (src.length > BLOCK_SRC_MAX || !DATA_IMG.test(src)) continue;
    const alt = typeof b.alt === 'string' ? b.alt.slice(0, 120) : '';
    const cap = typeof b.cap === 'string' ? b.cap.slice(0, 200) : '';
    out.push({ t: 'image', src, alt, ...(cap ? { cap } : {}) });
  }
  return out;
}

/** 不可枚举 token：32 字节 CSPRNG → base64url（43 字符，无公共前缀、不递增）。 */
export function newShareToken(): string {
  const b = new Uint8Array(32);
  crypto.getRandomValues(b);
  let s = '';
  for (const x of b) s += String.fromCharCode(x);
  return btoa(s).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
}

/**

 * 包括一个叫 `logo.quote.unitPrice` 的东西。所以子树也要过一遍键名对照。
 */
function clampDepth(v: unknown, depth: number, budget: { n: number }): unknown {
  if (depth <= 0 || budget.n <= 0) return null;
  budget.n--;
  if (Array.isArray(v)) return v.slice(0, 20).map((x) => clampDepth(x, depth - 1, budget));
  if (v && typeof v === 'object') {
    const o: Record<string, unknown> = {};
    // 🔴 **不能用 `Object.entries(v).slice(0, 30)`** —— 那会【先把整个对象枚举成数组】再截。
    //    节点预算管的是递归和输出，管不到这一步的分配：一个 100 万属性的对象
    //    实测约 589ms（Codex 抓到），而 body 上限 16MB 装得下非常多的短 key。
    //    `for...in` + 计数早退不分配那个中间数组。
    let seen = 0;
    for (const k in v as Record<string, unknown>) {
      if (!Object.prototype.hasOwnProperty.call(v, k)) continue;
      if (seen >= 30 || budget.n <= 0) break;
      seen++;
      if (COMMERCIAL_KEY.test(k) || IDENTITY_KEY.test(k)) continue;   // 子树里的商务字段就地丢掉
      o[k] = clampDepth((v as Record<string, unknown>)[k], depth - 1, budget);
    }
    return o;
  }
  if (typeof v === 'string') return v.slice(0, 200);
  return v;
}
