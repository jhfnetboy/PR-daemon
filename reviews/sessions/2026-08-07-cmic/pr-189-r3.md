def _span_energy(v, q=_Q, floor_pct=_FLOOR_PCT):
    """一维结构能量剖面 → (lo, hi, 跨度内/外能量密度比)。

    ⚠️ 阈值【相对噪声底】,不是相对 0 —— 旧版栽的就是这里(照片颗粒把每一列都垫过 12%-of-peak,
       跨度撑满整帧,量到的是画布)。步骤:平滑 → 扣分位数底噪 → 取累积能量中间 (1-2q) 段。
    """
    import numpy as np

    n = v.size
    if n < 16:
        return None
    k = max(3, (int(n * 0.01) | 1))                      # ~1% 宽的滑动平均,压掉逐列抖动
    # ⚠️ 必须【边缘复制】补边再卷积:np.convolve(mode="same") 是零填充,会把剖面两端人为压低,
    #    于是能量分位往里缩 —— 均匀能量下跨度只有 ~91% 而不是理论的 (1-2q)=94%,
    #    下面那条「满帧即退化」的判据就永远够不着(单测抓到过)。
    pad = k // 2
    sm = np.convolve(np.pad(v, pad, mode="edge"), np.ones(k) / k, mode="valid")
    excess = np.clip(sm - float(np.percentile(sm, floor_pct)), 0, None)
    total = float(excess.sum())
    if total <= 1e-6:                                    # 全帧无结构(纯色)→ 测不出
        return None
    cum = np.cumsum(excess) / total
    lo = int(np.searchsorted(cum, q))
    hi = min(int(np.searchsorted(cum, 1.0 - q)), n - 1)
    if hi <= lo:
        return None
    outside = np.concatenate([sm[:lo], sm[hi + 1:]])
    # 密度比只作【诊断】打进日志(它随颗粒强度塌向 1,不适合当判据),不参与放行/拦截决策。
    dens = float(sm[lo:hi + 1].mean()) / max(float(outside.mean()), 1e-9) if outside.size else float("inf")
    return lo, hi, dens


def measure(img):
    """估主体外接框。返回 dict 或 None(测不准 —— 调用方一律按放行处理)。

    盒子是画面主体,和背景之间必然有**结构**(边缘),所以用逐行/逐列的相邻像素差分强度做投影。
    比「按颜色分割背景」稳:CG 图背景是干净的纯色/渐变,增强图背景是照片场景,两者
    【没有共同的背景模型】,但「主体处结构强、背景处结构弱」在两边都成立 —— 前提是背景不太花,
    背景一花(精品店散景)就量到整个场景,见模块顶部。
    """
    import numpy as np

    g = np.asarray(img.convert("L"), dtype=np.float32)
    if g.size == 0 or g.shape[0] < 16 or g.shape[1] < 16:
        return None
    H, W = g.shape
    # 相邻差分 = 结构强度。轴向各做一次,避免只对横/竖边敏感。
    sx = _span_energy(np.abs(np.diff(g, axis=1)).sum(axis=0))   # 每列结构量 → 定水平范围
    sy = _span_energy(np.abs(np.diff(g, axis=0)).sum(axis=1))   # 每行结构量 → 定垂直范围
    if not sx or not sy:
        return None
    w = sx[1] - sx[0] + 1
    h = sy[1] - sy[0] + 1
    if w < 8 or h < 8:
        return None
    fill_x, fill_y = w / W, h / H
    return {"w": w, "h": h, "ratio": w / h, "fill_x": fill_x, "fill_y": fill_y,
            "dens_x": sx[2], "dens_y": sy[2],
            # 满帧 = 量到画布。此时 ratio 恒等于画布比例,好坏图无法区分 → 只能放弃判定。

/**
 * 对话里的图存哪儿 —— **不能存 localStorage**。
 *
 * ## 这个模块解决的是一个线上 bug
 *
 * 客户从确认页点「Back to chat」回来,对话在、**图全没了**。
 *
 * 根因:对话存档走 `localStorage`(配额 ~5MB),而照片级交付图是
 * `data:image/png;base64,…` —— **一张 1536×1024 就 2–4MB**。存不下时
 * `flushPersist` 会降级存「剥图版」(`im.remove()` 删掉所有 data: 图),
 * 于是对话保住了、图片一张不剩。那个降级本身是对的(保对话优先),
 * **错的是一开始就把图往 localStorage 里塞**。
 *
 * IndexedDB 的配额是按磁盘剩余空间算的(通常几百 MB 到几 GB),存图正合适。
 * 所以:**localStorage 只存 HTML + 图的 key,图本体进 IndexedDB。**
 *
 * ⚠️ 任何一步失败都**静默降级**,绝不让存图搞挂对话 —— 隐私模式/配额耗尽/
 *    浏览器不支持 IndexedDB 时,图拿不回来最多是"回来看不到图"(即今天的行为),
 *    而不是"对话也打不开"。
 */

const DB_NAME = 'cmic-chat-img';
const STORE = 'img';
const VERSION = 1;

let dbPromise: Promise<IDBDatabase | null> | null = null;

function openDb(): Promise<IDBDatabase | null> {
  if (dbPromise) return dbPromise;
  // ⚠️ **失败不缓存**:第一版把 null 也记进 `dbPromise`,于是一次瞬时故障
  //    (标签页竞争 / 临时 blocked)就让整个图层在本页生命周期内永久失效。
  const p: Promise<IDBDatabase | null> = new Promise((res) => {
    try {
      if (typeof indexedDB === 'undefined') return res(null);
      const req = indexedDB.open(DB_NAME, VERSION);
      req.onupgradeneeded = () => {
        const db = req.result;
        if (!db.objectStoreNames.contains(STORE)) db.createObjectStore(STORE);
      };
      req.onsuccess = () => res(req.result);
      req.onerror = () => res(null);
      req.onblocked = () => res(null);
    } catch { res(null); }
  });
  dbPromise = p.then((db) => { if (!db) dbPromise = null; return db; });
  return dbPromise;
}

function tx(db: IDBDatabase, mode: IDBTransactionMode): IDBObjectStore {
  return db.transaction(STORE, mode).objectStore(STORE);
}

/**
 * 一张图的 key —— 内容哈希 + 长度。
 *
 * ⚠️ **抽样哈希**(每 97 字节取一个),不是全扫:一张图 3MB,全扫要几毫秒 ×
 *    每次 persist × 每张图,会卡主线程。配合长度做后缀,碰撞概率低到可以忽略,
 *    而且哈希天然去重 —— 同一张图在多轮对话里出现,只存一份。
 */
export function imgKey(dataUrl: string): string {
  let h = 0x811c9dc5;
  for (let i = 0; i < dataUrl.length; i += 97) {
    h ^= dataUrl.charCodeAt(i);
    h = Math.imul(h, 0x01000193);
  }
  return `${(h >>> 0).toString(36)}-${dataUrl.length.toString(36)}`;
}

/** 写入若干张图。失败静默 —— 存图失败最多是回来看不到图,不该影响对话。 */
export async function putImages(entries: Array<[string, string]>): Promise<void> {
  if (!entries.length) return;
  const db = await openDb();
  if (!db) return;
  await new Promise<void>((res) => {
    try {
      const t = db.transaction(STORE, 'readwrite');
      const st = t.objectStore(STORE);
      for (const [k, v] of entries) st.put(v, k);
      t.oncomplete = () => res();
      t.onerror = () => res();
      t.onabort = () => res();     // 配额满会 abort —— 静默,别抛
    } catch { res(); }
  });
}

/** 取回若干张图。取不到的 key 直接不在返回里(调用方保留占位即可)。 */
export async function getImages(keys: string[]): Promise<Map<string, string>> {
  const out = new Map<string, string>();
  if (!keys.length) return out;
  const db = await openDb();
  if (!db) return out;
  await new Promise<void>((res) => {
    try {
      // 🔴 **事务级的 abort/error 必须也能 settle。**
      //    第一版只在每个请求的 onsuccess/onerror 里递减计数,而事务本身被丢弃 ——
      //    读事务一旦 abort,promise **永不 settle**,`restoreImages()`(await 它)
      //    永远不完成:所有图停在占位、`data-real` 还挂着,正好喂给 extractImages 那个 bug。
      const t = db.transaction(STORE, 'readonly');
      const st = t.objectStore(STORE);
      let settled = false;
      const finish = () => { if (!settled) { settled = true; res(); } };
      t.oncomplete = finish; t.onerror = finish; t.onabort = finish;
      let left = keys.length;
      const done = () => { if (--left <= 0) finish(); };
      for (const k of keys) {
        const r = st.get(k);
        r.onsuccess = () => { if (typeof r.result === 'string') out.set(k, r.result); done(); };
        r.onerror = () => done();
      }
    } catch { res(); }
  });
  return out;
}

/**
 * 清掉不再被任何存档引用的图。
 *
 * ⚠️ 不做这一步的话,客户每换一次颜色就多一张永远不会被读到的图躺在磁盘上 ——
 *    IndexedDB 配额大,但不是无限。
 */
export async function pruneExcept(keep: string[], allKeysInUse?: string[]): Promise<void> {
  const db = await openDb();
  if (!db) return;
  // ⚠️ **对象仓库是 origin 全局的,而对话存档是【按用户】分键的**(`cmic.chat.v1:<id>`)。
  //    只按"当前这份存档"去 prune,会删掉**同一浏览器上其他账号**存档仍在引用的图本体 ——
  //    共享电脑上 A 开个新对话就能让 B 的图全没(而匿名出图根本不落 R2,那就是永久消失)。
  //    所以调用方要把「所有存档里仍在用的 key」一起传进来。
  const keepSet = new Set([...keep, ...(allKeysInUse ?? [])]);
  await new Promise<void>((res) => {
    try {
      const t = db.transaction(STORE, 'readwrite');
      const st = t.objectStore(STORE);
      const req = st.getAllKeys();
      req.onsuccess = () => {
        for (const k of req.result) if (typeof k === 'string' && !keepSet.has(k)) st.delete(k);
      };
      t.oncomplete = () => res();
      t.onerror = () => res();
      t.onabort = () => res();
    } catch { res(); }
  });
}

  const db = await openDb();
  if (!db) return;

    try {
      const t = db.transaction(STORE, 'readwrite');
      const st = t.objectStore(STORE);
      for (const k of keys) st.delete(k);
      t.oncomplete = () => res();
      t.onerror = () => res();
      t.onabort = () => res();
    } catch { res(); }
  });
}

/** Uint8Array → base64(分块避免大数组爆栈) */
function bytesToB64(bytes: Uint8Array): string {
  let s = '';
  const chunk = 0x8000;
  for (let i = 0; i < bytes.length; i += chunk) {

}

 *  tint = 工艺色:用 logo 的 alpha 当 mask 填色,烫金→金、印黑→黑(和 WebGL 一致)。不给则保留原色。 */
function logoCutB64(logo: HTMLImageElement, tint?: string): string {
  const w = logo.naturalWidth || 600;
  const h = logo.naturalHeight || 160;
  const cv = document.createElement('canvas');
  cv.width = w; cv.height = h;
  const g = cv.getContext('2d')!;
  g.drawImage(logo, 0, 0, w, h);
  if (tint) {
    g.globalCompositeOperation = 'source-in'; // 保 logo 形状(alpha),整体上工艺色
    g.fillStyle = tint; g.fillRect(0, 0, w, h);
  }
  return canvasB64(cv);
}

/** logo 所在面的 4 个世界坐标角点(Three.js 坐标,mm)。天地盒/找不到 → null(走 bbox 顶面老路径) */
export function logoFaceQuad(faces: Face3D[], boxType: string): number[] | null {
  const id = LOGO_FACE_ID[boxType];
  const f = id ? faces.find((x) => (x as { id?: string }).id === id) : undefined;
  if (!f || f.corners.length < 4) return null;
  return f.corners.slice(0, 4).flatMap((c) => [c[0], c[1], c[2]]); // 12 个数
}

/** 盒子实际外形尺寸(Three.js:x=宽 y=高 z=深,mm)—— 送后台让 AI 保持这个比例,别把长条画成方盒 */
function facesBBox(faces: Face3D[]): [number, number, number] {
  const mn = [1e9, 1e9, 1e9], mx = [-1e9, -1e9, -1e9];
  for (const f of faces) for (const c of f.corners) for (let i = 0; i < 3; i++) {
    mn[i] = Math.min(mn[i]!, c[i]!); mx[i] = Math.max(mx[i]!, c[i]!);
  }
  return [mx[0]! - mn[0]!, mx[1]! - mn[1]!, mx[2]! - mn[2]!];
}
